#include <windows.h>
#include <stdio.h>

int main(int argc, char *argv[])
{

HANDLE hMapFile;

LPVOID lpMapAddress;

hMapFile = OpenFileMapping (FILE MAP ALL ACCESS, /* R/W access x/
FALSE, /* no inheritance */         .
TEXT ("SharedObject")); /* name of mapped file object */

1pMapAddress = MapViewOfFile(hMapFile, /* mapped object handle x/
FILE MAP ALL ACCESS, /* read/write access */
0, /* mapped view of entire file */
0,
0);

/* read from shared memory */
printf ("Read message %s", lpMapAddress);

UnmapViewOfFile (1pMapAddress) ;
CloseHandle (hMapFile) ;

}
