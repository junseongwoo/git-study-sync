, C.TEAM_NAME, D.TEAM_NAME
, A.HOME_SCORE, A.AWAY_SCORE
FROM SCHEDULE A JOIN STADIUM B
ON B.STADIUM_ID = A,STADIUM_ID
JOIN TEAM C
ON C.TEAM_ID = A.HOMETEAM_ID

ON D.TEAM_ID = A AWAYTEAM_ID
