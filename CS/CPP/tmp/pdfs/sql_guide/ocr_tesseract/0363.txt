예제를 위해 다음과 같이 테이블을 생성하자.

DROP TABLE 11 PURGE;
CREATE TABLE T1 AS
SELECT 308, 010 54, D28_SAL, 010 01, D26_CNT

FROM (SELECT 308, DEPTNO, SAL FROM EMP WHERE JOB IN ('ANALYST', 'CLERK'))
PIVOT (SUM (SAL) AS SAL, COUNT (*) AS CNT FOR DEPTNO IN (10 AS 010, 20 AS 020) );

FROM T1
UNPIVOT (SAL FOR DEPTNO IN (D16_SAL, D26_SAL))
