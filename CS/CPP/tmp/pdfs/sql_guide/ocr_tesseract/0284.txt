[예제] Oracle

SELECT PLAYER NAME AS 선수명, POSITION AS 포지션
, 840 \0 AS 백넘버, HEIGHT AS 7|
FROM (SELECT ~ PLAYER_NAME, POSITION, BACK_NO, HEIGHT
FROM PLAYER
WHERE HEIGHT IS NOT NULL
ORDER BY HEIGHT DESC)
WHERE ROWNUM <= 5;
