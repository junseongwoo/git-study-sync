SELECT '7' AS 구분코드, POSITION AS 포지션, ROUND (AVG (HEIGHT), 3) AS 평균키
FROM PLAYER

GROUP BY POSITION

UNION ALL

SELECT 'T' AS 구분코드, TEALID AS 팀명, ROUND (AVG (HEIGHT), 3) AS 평균키
FROM PLAYER

GROUP BY TEAM_ID

ORDER BY 1;
