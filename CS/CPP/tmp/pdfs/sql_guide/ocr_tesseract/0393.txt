INTO TEAM (TEAM_ID, REGION_NAME, TEAM_NAME, ORIG_YYYY, STADIUM_ID)
SELECT REPLACE(TEAM_ID, 'K', 'A') AS TEAM_ID
, REGION_NAME, REGION NAME |} ' SAE} AS TEAM_NAME
, 2819 AS ORIG_YYYY, STADIUM_ID
