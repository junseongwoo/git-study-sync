SELECT 절에 부여된 열 명을 지정하면 필요한 열만 조회할 수 있다.

SELECT . 208, D26_SAL
FROM (SELECT 208, DEPTNO, SAL FROM EWP)
PIVOT (SUM (SAL) AS SAL FOR DEPTNO IN (10 AS 010, 20 AS 020, 30 AS D38))
WHERE D20_SAL > 2500

ORDER BY 1;
