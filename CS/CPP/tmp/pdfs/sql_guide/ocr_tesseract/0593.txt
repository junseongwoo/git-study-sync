oli Oracle SQL Server 각각에서 해시 조인으로 유도했을 때의 실행계획이다.

select /*+ ordered use_hash(e) */ d.deptno, d.dname, e.empno, e.ename
from dept d, emp e
where d.deptno = e.deptno
