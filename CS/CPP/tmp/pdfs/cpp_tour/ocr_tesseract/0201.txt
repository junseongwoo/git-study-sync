([[:alpha:]]|_)[[:alnum:]]*       // 틀림: 밑줄은 3ㄴ040에 속하지 않음

([[:alpha:]1]|_)([[:alnum:]]|_)*    // OK: 그러나 지저분함

[[:alpha:]_][[:alnum:]_]*         // OK: 문자 클래스 안에 밀줄이 있음
[_[:alpha:]]1[_[:alnum:]]*          // OK
[_[:alpha:]]\w*                   // \we [_[:alnum:]]2 동일

마지막으로 주어진 문자열이 식별자인지 확인하기 위해 가장 간단한 BA 2] regex
match( )(9.4.1절)를 사용하는 함수를 살펴보자.

bool is_identifier(const string& 5)
{
regex pat {"[_[:alpha:]]\\w?"}; // 밑줄이나 알파벳 하나가 오고,
// 그 뒤로 0개 이상의 밀줄이나 알파벳, 숫자가 옴
return regex_match(s,pat);

}

일반 문자열 리터럴에서 백슬래시를 포함시키려면 백슬래시를 두 번 써야 한다는 점
을 기억하자. 로우 문자열 리터럴을 이용하면 이러한 문제를 덜 수 있다.

bool is_identifier(const string& 5)
{
regex pat {R"([_[:alpha:]]\w?)"};
return regex_match(s,pat);
}
