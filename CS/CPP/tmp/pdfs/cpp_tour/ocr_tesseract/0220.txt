template<typename Target =string, typename Source =string>
Target to(Source arg)      // Source Targeto@ 변환
{

stringstream interpreter;

Target result;

if (!(interpreter << arg)                 // arg 스트림에 출력
|| !(interpreter >> result)          // AER0IM result 읽기
|| !(interpreter >> std::ws).eof()) // 스트림에 남은 것이 있는가?
throw runtime_error{"to<() failed"};

return result;

}

함수 템플릿 인자를 FEU 수 없거나 기본 값이 존재하지 않는 경우에만 명시적으로
지정해야 한다(7.2.4절), 따라서 다음과 같이 작성할 수 있다.

auto x1 = to<string,double>(1.2); // 매우 명시적(그리고 코드가 길어짐)
auto x2 = to<string>(1.2);          // Sources double2 추론됨
auto x3 = to<(1.2); // Target? 7|2422 string; Source double2 22%
auto x4 = to(1.2);     // the 도 280/22 필요 없음
// Target® 7|2422 string; Source double2 추론됨

모든 함수 템플릿 인자에 기본 값을 적용하면 <>를 생략할 수 있다.

이 예제를 바탕으로 언어 기능과 표준 라이브러리 기능을 함께 사용해 일반성과 사용
의 편리성을 달성하는 방법을 살펴볼 수 있다.
