// user.cpp:

#include "Vector.h" // Vector? 인터페이스 가져오기
#include <cmath>     // sqrt()& 포함하는 표준 라이브러리 수학 함수 인터페이스 가쳐오기

double sqrt_sum(Vector& v)
{
double sum = 0;
for (int i=0; i!=v.size(); +1)
sum=std::sqrt(v[i]); // 제곱근의 합
return sum;

}
